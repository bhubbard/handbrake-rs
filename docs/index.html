Rust Programmatic API
use handbrake_rs::{TranscodeEngine, TranscodeJob, Preset};
use std::path::PathBuf;

fn main() -> anyhow::Result<()> {
    let preset = Preset::find("Fast 1080p30")?;
    let job = TranscodeJob::from_preset(
        PathBuf::from("raw.mkv"),
        PathBuf::from("dist.mp4"),
        &preset,
    );

    TranscodeEngine::run(&job, |prog| {
        println!("{:.1}% | {:.1} fps | ETA: {}s", 
            prog.percent, prog.fps, prog.eta_seconds);
    })?;
    Ok(())
}
